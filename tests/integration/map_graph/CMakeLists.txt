cmake_minimum_required(VERSION 3.16)
project(drift_map_graph_probes LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(slam_toolbox REQUIRED)
find_package(nav2_map_server REQUIRED)
find_package(Ceres REQUIRED)
find_package(Boost REQUIRED COMPONENTS serialization thread)
find_package(TBB REQUIRED)
find_package(pluginlib REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(SLAM_SOURCE "${ROOT}/.cache/slam_backend/src/slam_toolbox" CACHE PATH "Prepared SLAM source")
set(KARTO_LIBRARY "" CACHE FILEPATH "Optional explicit library for an uninstalled backend test build")
if(NOT KARTO_LIBRARY)
  find_library(INSTALLED_KARTO_LIBRARY NAMES kartoSlamToolbox
    HINTS "${slam_toolbox_DIR}/../../../lib" NO_DEFAULT_PATH REQUIRED)
  set(KARTO_LIBRARY "${INSTALLED_KARTO_LIBRARY}")
endif()
add_executable(metadata_roundtrip metadata_roundtrip.cpp)
target_include_directories(metadata_roundtrip BEFORE PRIVATE "${SLAM_SOURCE}/lib/karto_sdk/include")
ament_target_dependencies(metadata_roundtrip rclcpp nav_msgs nav2_map_server slam_toolbox)
target_link_libraries(metadata_roundtrip "${KARTO_LIBRARY}" Boost::serialization Boost::thread TBB::tbb)
add_executable(graph_diagnostics graph_diagnostics.cpp "${SLAM_SOURCE}/solvers/ceres_solver.cpp")
target_include_directories(graph_diagnostics BEFORE PRIVATE "${SLAM_SOURCE}" "${SLAM_SOURCE}/lib/karto_sdk/include" "${SLAM_SOURCE}/include")
ament_target_dependencies(graph_diagnostics rclcpp slam_toolbox pluginlib)
target_link_libraries(graph_diagnostics "${KARTO_LIBRARY}" Ceres::ceres Boost::serialization Boost::thread TBB::tbb)
add_executable(reference_reparameterization reference_reparameterization.cpp)
target_include_directories(reference_reparameterization BEFORE PRIVATE "${SLAM_SOURCE}")
target_link_libraries(reference_reparameterization Ceres::ceres)

add_executable(transaction_probe transaction_probe.cpp "${SLAM_SOURCE}/solvers/ceres_solver.cpp")
target_include_directories(transaction_probe BEFORE PRIVATE "${SLAM_SOURCE}" "${SLAM_SOURCE}/lib/karto_sdk/include" "${SLAM_SOURCE}/include")
ament_target_dependencies(transaction_probe rclcpp slam_toolbox pluginlib)
target_link_libraries(transaction_probe "${KARTO_LIBRARY}" Ceres::ceres Boost::serialization Boost::thread TBB::tbb)
add_executable(evidence_archive evidence_archive.cpp)
target_include_directories(evidence_archive BEFORE PRIVATE "${SLAM_SOURCE}" "${SLAM_SOURCE}/lib/karto_sdk/include" "${SLAM_SOURCE}/include")
ament_target_dependencies(evidence_archive rclcpp slam_toolbox)
target_link_libraries(evidence_archive "${KARTO_LIBRARY}" Boost::serialization Boost::thread TBB::tbb)
