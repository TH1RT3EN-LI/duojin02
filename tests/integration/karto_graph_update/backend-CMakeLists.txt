cmake_minimum_required(VERSION 3.16)
project(karto_backend_prepared_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
find_package(Ceres REQUIRED)
find_package(Boost REQUIRED COMPONENTS serialization thread filesystem system)
find_package(TBB REQUIRED)
find_package(pluginlib REQUIRED)
set(ROOT /work)
set(SLAM_SOURCE "${ROOT}/.cache/slam_backend/graph-update-stage" CACHE PATH "Prepared backend source")
add_executable(backend_probe "${ROOT}/tests/integration/karto_graph_update/backend_probe.cpp" "${SLAM_SOURCE}/lib/karto_sdk/src/Karto.cpp" "${SLAM_SOURCE}/lib/karto_sdk/src/Mapper.cpp" "${SLAM_SOURCE}/solvers/ceres_solver.cpp")
target_include_directories(backend_probe BEFORE PRIVATE "${ROOT}/src/duojin02_scan_geometry/include" "${SLAM_SOURCE}" "${SLAM_SOURCE}/lib/karto_sdk/include" "${SLAM_SOURCE}/include")
ament_target_dependencies(backend_probe rclcpp std_srvs pluginlib)
target_link_libraries(backend_probe Ceres::ceres Boost::serialization Boost::thread Boost::filesystem Boost::system TBB::tbb)
target_compile_options(backend_probe PRIVATE -O1 -g0 -fsanitize=undefined,float-cast-overflow -fno-sanitize-recover=all)
target_link_options(backend_probe PRIVATE -fsanitize=undefined,float-cast-overflow -fno-sanitize-recover=all)
