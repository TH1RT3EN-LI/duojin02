cmake_minimum_required(VERSION 3.16)
project(snapshot_ros_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(slam_toolbox REQUIRED)
find_package(nav2_map_server REQUIRED)
find_package(pluginlib REQUIRED)
find_package(Ceres REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(TBB REQUIRED)
find_package(Boost REQUIRED COMPONENTS serialization thread filesystem system)
set(SNAPSHOT_REPO /work CACHE PATH "GPU repository")
set(SLAM_SOURCE "${SNAPSHOT_REPO}/.cache/slam_backend/graph-update-stage" CACHE PATH "Current shared source stage")
set(TOOLBOX_BUILD /results/snapshot/toolbox-build CACHE PATH "Exact private development build")
set(COMPONENT_DIR "${TOOLBOX_BUILD}")
if(EXISTS "${TOOLBOX_BUILD}/lib/libtoolbox_common.so")
  set(COMPONENT_DIR "${TOOLBOX_BUILD}/lib")
endif()
find_library(SNAPSHOT_KARTO kartoSlamToolbox PATHS "${COMPONENT_DIR}/lib/karto_sdk" "${COMPONENT_DIR}" NO_DEFAULT_PATH REQUIRED)
add_executable(snapshot_ros_probe "${SNAPSHOT_REPO}/tests/integration/karto_graph_update/snapshot_ros_probe.cpp")
target_compile_definitions(snapshot_ros_probe PRIVATE SNAPSHOT_WIRE_CLIENT_PATH="${SNAPSHOT_REPO}/tests/integration/karto_graph_update/snapshot_load_wire_client.py")
target_include_directories(snapshot_ros_probe BEFORE PRIVATE "${TOOLBOX_BUILD}/rosidl_generator_cpp" "${SNAPSHOT_REPO}/src/duojin02_scan_geometry/include" "${SLAM_SOURCE}" "${SLAM_SOURCE}/include" "${SLAM_SOURCE}/lib/karto_sdk/include")
ament_target_dependencies(snapshot_ros_probe rclcpp slam_toolbox nav2_map_server pluginlib)
target_link_libraries(snapshot_ros_probe "${COMPONENT_DIR}/libtoolbox_common.so" "${COMPONENT_DIR}/libceres_solver_plugin.so" "${SNAPSHOT_KARTO}" "${COMPONENT_DIR}/libslam_toolbox__rosidl_typesupport_cpp.so" Ceres::ceres OpenSSL::Crypto TBB::tbb Boost::serialization Boost::thread Boost::filesystem Boost::system)
