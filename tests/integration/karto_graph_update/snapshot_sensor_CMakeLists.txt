cmake_minimum_required(VERSION 3.16)
project(snapshot_sensor_lifecycle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Boost REQUIRED COMPONENTS serialization filesystem system)
find_package(TBB REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(SNAPSHOT_REPO "${REPO}" CACHE PATH "GPU repository")
set(SLAM_SOURCE "${REPO}/.cache/slam_backend/graph-update-stage" CACHE PATH "Stage source under test")
add_executable(snapshot_sensor_lifecycle snapshot_sensor_lifecycle.cpp "${SLAM_SOURCE}/lib/karto_sdk/src/Karto.cpp")
target_include_directories(snapshot_sensor_lifecycle BEFORE PRIVATE "${SNAPSHOT_REPO}/src/duojin02_scan_geometry/include" "${SLAM_SOURCE}/lib/karto_sdk/include")
target_link_libraries(snapshot_sensor_lifecycle Boost::serialization Boost::filesystem Boost::system TBB::tbb)
target_compile_options(snapshot_sensor_lifecycle PRIVATE -O1 -g0 -fsanitize=undefined,float-cast-overflow -fno-sanitize-recover=all)
target_link_options(snapshot_sensor_lifecycle PRIVATE -fsanitize=undefined,float-cast-overflow -fno-sanitize-recover=all)
