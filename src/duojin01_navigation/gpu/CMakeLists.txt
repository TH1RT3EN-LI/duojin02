cmake_minimum_required(VERSION 3.18)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(duojin_gpu_components LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  include(CTest)
endif()
option(DUOJIN_ENABLE_CUDA "Build optional CUDA navigation backend" OFF)
add_library(duojin_gpu_backend SHARED gpu_backend.cpp gpu_sampling.cpp)
target_include_directories(duojin_gpu_backend PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/../include>
  $<INSTALL_INTERFACE:include>)
target_compile_features(duojin_gpu_backend PUBLIC cxx_std_17)
if(DUOJIN_ENABLE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 87 CACHE STRING "CUDA architectures; 87 for Orin, 120 for RTX 5070")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  target_sources(duojin_gpu_backend PRIVATE gpu_backend.cu)
  target_compile_definitions(duojin_gpu_backend PRIVATE DUOJIN_HAS_CUDA=1)
  target_compile_options(duojin_gpu_backend PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
  set_target_properties(duojin_gpu_backend PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_link_libraries(duojin_gpu_backend PRIVATE CUDA::cudart_static)
endif()
install(TARGETS duojin_gpu_backend LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  add_executable(gpu_sampling_checks sampling_checks.cpp)
  target_link_libraries(gpu_sampling_checks duojin_gpu_backend)
  target_compile_options(gpu_sampling_checks PRIVATE -UNDEBUG)
  add_test(NAME gpu_sampling_checks COMMAND gpu_sampling_checks)
  add_executable(gpu_geometry_checks geometry_checks.cpp)
  target_link_libraries(gpu_geometry_checks duojin_gpu_backend)
  target_compile_options(gpu_geometry_checks PRIVATE -UNDEBUG)
  add_test(NAME gpu_geometry_checks COMMAND gpu_geometry_checks)
  if(DUOJIN_ENABLE_CUDA)
    add_test(NAME gpu_sampling_cuda_checks COMMAND gpu_sampling_checks --require-cuda)
    set_tests_properties(gpu_sampling_cuda_checks PROPERTIES SKIP_RETURN_CODE 77)
    add_test(NAME gpu_sampling_no_device_checks COMMAND gpu_sampling_checks)
    set_tests_properties(gpu_sampling_no_device_checks PROPERTIES ENVIRONMENT "CUDA_VISIBLE_DEVICES=")
    add_test(NAME gpu_geometry_cuda_checks COMMAND gpu_geometry_checks --require-cuda)
    set_tests_properties(gpu_geometry_cuda_checks PROPERTIES SKIP_RETURN_CODE 77)
    add_test(NAME gpu_geometry_no_device_checks COMMAND gpu_geometry_checks)
    set_tests_properties(gpu_geometry_no_device_checks PROPERTIES ENVIRONMENT "CUDA_VISIBLE_DEVICES=")
  endif()
  add_executable(gpu_geometry_benchmark EXCLUDE_FROM_ALL
    "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/performance/gpu_geometry_benchmark.cpp")
  target_link_libraries(gpu_geometry_benchmark duojin_gpu_backend)
  add_executable(gpu_sampling_benchmark EXCLUDE_FROM_ALL
    "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/performance/gpu_sampling_benchmark.cpp")
  target_link_libraries(gpu_sampling_benchmark duojin_gpu_backend)
  add_executable(gpu_structured_benchmark EXCLUDE_FROM_ALL
    "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/performance/structured_sampling_benchmark.cpp")
  target_compile_definitions(gpu_structured_benchmark PRIVATE DUOJIN_STRUCTURED_API=1)
  target_link_libraries(gpu_structured_benchmark duojin_gpu_backend)
endif()
