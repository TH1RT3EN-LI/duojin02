cmake_minimum_required(VERSION 3.16)
project(karto_association_lifecycle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(slam_toolbox REQUIRED)
find_package(Ceres REQUIRED)
find_package(Boost REQUIRED COMPONENTS serialization thread filesystem system)
find_package(TBB REQUIRED)
find_package(pluginlib REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(SLAM_SOURCE "${ROOT}/.cache/slam_backend/src/slam_toolbox" CACHE PATH "Prepared source under test")
add_executable(lifecycle_probe lifecycle_probe.cpp "${SLAM_SOURCE}/lib/karto_sdk/src/Karto.cpp" "${SLAM_SOURCE}/solvers/ceres_solver.cpp")
# Include exact Mapper.cpp in the probe TU: manager accessors are defined inline
# in that implementation. This still runs the complete production implementation.
target_compile_definitions(lifecycle_probe PRIVATE KARTO_MAPPER_IMPLEMENTATION="${SLAM_SOURCE}/lib/karto_sdk/src/Mapper.cpp")
target_include_directories(lifecycle_probe BEFORE PRIVATE "${ROOT}/src/duojin02_scan_geometry/include" "${SLAM_SOURCE}" "${SLAM_SOURCE}/lib/karto_sdk/include" "${SLAM_SOURCE}/include")
ament_target_dependencies(lifecycle_probe rclcpp slam_toolbox pluginlib)
target_link_libraries(lifecycle_probe Ceres::ceres Boost::serialization Boost::thread Boost::filesystem Boost::system TBB::tbb)
target_compile_options(lifecycle_probe PRIVATE -O1 -g0 -fsanitize=undefined,float-cast-overflow -fno-sanitize-recover=all)
target_link_options(lifecycle_probe PRIVATE -fsanitize=undefined,float-cast-overflow -fno-sanitize-recover=all)
