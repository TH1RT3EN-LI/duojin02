cmake_minimum_required(VERSION 3.18)
project(duojin02_mppi_controller LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(xtensor REQUIRED)
find_package(xsimd REQUIRED)
set(dependencies_pkgs rclcpp rclcpp_lifecycle rcl_interfaces nav2_common pluginlib tf2
  geometry_msgs visualization_msgs nav_msgs nav2_core nav2_costmap_2d nav2_util
  tf2_geometry_msgs tf2_eigen tf2_ros)
foreach(pkg IN LISTS dependencies_pkgs)
  find_package(${pkg} REQUIRED)
endforeach()
option(DUOJIN_ENABLE_CUDA "Build CUDA MPPI optimizer" ON)
option(DUOJIN_CUDA_REQUIRED "Require CUDA compiler and device tests" OFF)
if(DUOJIN_CUDA_REQUIRED AND NOT DUOJIN_ENABLE_CUDA)
  message(FATAL_ERROR "DUOJIN_CUDA_REQUIRED requires DUOJIN_ENABLE_CUDA")
endif()
add_library(duojin02_mppi_cuda SHARED src/cuda/normal.cpp)
target_include_directories(duojin02_mppi_cuda PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
set(DUOJIN_MPPI_CUDA_BUILT OFF)
if(DUOJIN_ENABLE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    if(CMAKE_VERSION VERSION_GREATER_EQUAL 3.24)
      set(CMAKE_CUDA_ARCHITECTURES native CACHE STRING "CUDA architectures")
    else()
      set(CMAKE_CUDA_ARCHITECTURES 120 CACHE STRING "CUDA architectures")
    endif()
  endif()
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    target_sources(duojin02_mppi_cuda PRIVATE src/cuda/optimizer.cu)
    target_link_libraries(duojin02_mppi_cuda PRIVATE CUDA::cudart_static)
    target_compile_options(duojin02_mppi_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--ftz=false;--prec-div=true;--fmad=false>)
    set_target_properties(duojin02_mppi_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES}")
    set(DUOJIN_MPPI_CUDA_BUILT ON)
  elseif(DUOJIN_CUDA_REQUIRED)
    message(FATAL_ERROR "CUDA compiler required for MPPI")
  endif()
endif()
if(NOT DUOJIN_MPPI_CUDA_BUILT)
  target_sources(duojin02_mppi_cuda PRIVATE src/cuda/stub.cpp)
endif()
add_library(duojin02_mppi_controller_lib SHARED src/controller.cpp src/optimizer.cpp
  src/critic_manager.cpp src/trajectory_visualizer.cpp src/path_handler.cpp
  src/parameters_handler.cpp src/noise_generator.cpp)
file(GLOB critic_sources CONFIGURE_DEPENDS src/critics/*.cpp)
add_library(duojin02_mppi_critics SHARED ${critic_sources})
set(libraries duojin02_mppi_controller_lib duojin02_mppi_critics duojin02_mppi_cuda)
# Retain upstream CPU SIMD capability without applying x86 flags to nvcc or
# enabling fast-math (which would erase the finite-data safety contract).
include(CheckCXXCompilerFlag)
set(MPPI_CPU_ISA_FLAGS)
foreach(flag -mno-avx512f -msse4.2 -mavx2 -mfma)
  string(MAKE_C_IDENTIFIER "MPPI_SUPPORTS_${flag}" flag_variable)
  check_cxx_compiler_flag("${flag}" ${flag_variable})
  if(${flag_variable})
    list(APPEND MPPI_CPU_ISA_FLAGS "${flag}")
  endif()
endforeach()
foreach(lib duojin02_mppi_controller_lib duojin02_mppi_critics)
  target_include_directories(${lib} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> ${xsimd_INCLUDE_DIRS})
  target_compile_definitions(${lib} PUBLIC XTENSOR_ENABLE_XSIMD XTENSOR_USE_XSIMD)
  target_compile_options(${lib} PRIVATE -O3 -fno-fast-math -ffp-contract=off)
  foreach(flag IN LISTS MPPI_CPU_ISA_FLAGS)
    target_compile_options(${lib} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:${flag}>")
  endforeach()
  target_link_libraries(${lib} xtensor xtensor::use_xsimd duojin02_mppi_cuda)
  ament_target_dependencies(${lib} ${dependencies_pkgs})
  set_target_properties(${lib} PROPERTIES INSTALL_RPATH "$ORIGIN")
endforeach()
install(TARGETS ${libraries} ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(FILES UPSTREAM.json README.md LICENSE LICENSE.md LICENSE.Apache-2.0
  DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(mppi_core_checks test/core_checks.cpp)
  target_compile_options(mppi_core_checks PRIVATE -UNDEBUG -fno-fast-math -ffp-contract=off)
  target_link_libraries(mppi_core_checks duojin02_mppi_controller_lib)
  ament_target_dependencies(mppi_core_checks ${dependencies_pkgs})
  add_test(NAME mppi_core_cpu_checks COMMAND mppi_core_checks --cpu)
  if(DUOJIN_MPPI_CUDA_BUILT)
    add_test(NAME mppi_core_cuda_checks COMMAND mppi_core_checks --cuda)
    if(NOT DUOJIN_CUDA_REQUIRED)
      set_tests_properties(mppi_core_cuda_checks PROPERTIES SKIP_RETURN_CODE 77)
    endif()
  endif()
  add_test(NAME mppi_core_no_device_checks COMMAND mppi_core_checks --no-device)
  set_tests_properties(mppi_core_no_device_checks PROPERTIES ENVIRONMENT "CUDA_VISIBLE_DEVICES=")
endif()
ament_export_libraries(${libraries})
ament_export_dependencies(${dependencies_pkgs} xtensor xsimd)
ament_export_include_directories(include)
pluginlib_export_plugin_description_file(nav2_core mppic.xml)
pluginlib_export_plugin_description_file(duojin02_mppi_controller critics.xml)
ament_package()
