cmake_minimum_required(VERSION 3.16)
project(duojin02_scan_geometry)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_export_dependencies(rclcpp sensor_msgs tf2_msgs)
if(BUILD_TESTING)
  enable_testing()
  add_executable(scan_geometry_checks test/geometry_checks.cpp)
  target_compile_features(scan_geometry_checks PRIVATE cxx_std_17)
  target_include_directories(scan_geometry_checks PRIVATE include)
  add_test(NAME scan_geometry_checks COMMAND scan_geometry_checks)
endif()
ament_package()
