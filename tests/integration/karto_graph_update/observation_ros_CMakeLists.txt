cmake_minimum_required(VERSION 3.16)
project(observation_ros_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(slam_toolbox REQUIRED)
find_package(nav2_map_server REQUIRED)
find_package(pluginlib REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(Ceres REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(TBB REQUIRED)
find_package(Boost REQUIRED COMPONENTS serialization thread filesystem system)
set(SNAPSHOT_REPO /work CACHE PATH "GPU repository")
set(SLAM_SOURCE "${SNAPSHOT_REPO}/.cache/slam_backend/graph-update-throughput-stage" CACHE PATH "Current isolated stage")
set(TOOLBOX_BUILD "${SNAPSHOT_REPO}/output/lilocbench-graph-update-throughput-20261017/observation/toolbox-build" CACHE PATH "Private development build or exact installed prefix")
set(COMPONENT_DIR "${TOOLBOX_BUILD}")
if(EXISTS "${TOOLBOX_BUILD}/lib/libtoolbox_common.so")
  set(COMPONENT_DIR "${TOOLBOX_BUILD}/lib")
endif()
find_library(SNAPSHOT_KARTO kartoSlamToolbox PATHS "${COMPONENT_DIR}/lib/karto_sdk" "${COMPONENT_DIR}" NO_DEFAULT_PATH REQUIRED)
add_executable(observation_ros_probe "${SNAPSHOT_REPO}/tests/integration/karto_graph_update/observation_ros_probe.cpp")
target_include_directories(observation_ros_probe BEFORE PRIVATE "${TOOLBOX_BUILD}/rosidl_generator_cpp" "${SNAPSHOT_REPO}/src/duojin02_scan_geometry/include" "${SLAM_SOURCE}" "${SLAM_SOURCE}/include" "${SLAM_SOURCE}/lib/karto_sdk/include")
ament_target_dependencies(observation_ros_probe rclcpp slam_toolbox nav2_map_server pluginlib)
target_link_libraries(observation_ros_probe "${COMPONENT_DIR}/libtoolbox_common.so" "${COMPONENT_DIR}/libsync_slam_toolbox.so" "${COMPONENT_DIR}/libasync_slam_toolbox.so" "${COMPONENT_DIR}/liblocalization_slam_toolbox.so" "${COMPONENT_DIR}/liblifelong_slam_toolbox.so" "${COMPONENT_DIR}/libmap_and_localization_slam_toolbox.so" "${COMPONENT_DIR}/libceres_solver_plugin.so" "${SNAPSHOT_KARTO}" "${COMPONENT_DIR}/libslam_toolbox__rosidl_typesupport_cpp.so" Ceres::ceres OpenSSL::Crypto TBB::tbb Boost::serialization Boost::thread Boost::filesystem Boost::system)

add_executable(laser_inversion_probe "${SNAPSHOT_REPO}/tests/integration/karto_graph_update/laser_inversion_probe.cpp" "${SLAM_SOURCE}/src/laser_utils.cpp")
target_include_directories(laser_inversion_probe BEFORE PRIVATE "${SNAPSHOT_REPO}/src/duojin02_scan_geometry/include" "${SLAM_SOURCE}/include" "${SLAM_SOURCE}/lib/karto_sdk/include")
ament_target_dependencies(laser_inversion_probe rclcpp slam_toolbox sensor_msgs tf2_ros tf2_geometry_msgs)
target_link_libraries(laser_inversion_probe "${SNAPSHOT_KARTO}" TBB::tbb Boost::serialization Boost::filesystem Boost::system)
target_compile_options(laser_inversion_probe PRIVATE -O1 -g0 -fsanitize=address,undefined -fno-sanitize-recover=all)
target_link_options(laser_inversion_probe PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
