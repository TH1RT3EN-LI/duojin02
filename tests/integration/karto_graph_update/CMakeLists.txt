cmake_minimum_required(VERSION 3.16)
project(karto_graph_update LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
find_package(Ceres REQUIRED)
find_package(Boost REQUIRED COMPONENTS serialization thread filesystem system)
find_package(TBB REQUIRED)
find_package(pluginlib REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(SLAM_SOURCE "${ROOT}/.cache/slam_backend/src/slam_toolbox" CACHE PATH "Prepared production source under test")
add_executable(coordinator_probe coordinator_probe.cpp "${SLAM_SOURCE}/lib/karto_sdk/src/Karto.cpp" "${SLAM_SOURCE}/solvers/ceres_solver.cpp")
target_compile_definitions(coordinator_probe PRIVATE KARTO_MAPPER_IMPLEMENTATION="${SLAM_SOURCE}/lib/karto_sdk/src/Mapper.cpp")
target_include_directories(coordinator_probe BEFORE PRIVATE "${ROOT}/src/duojin02_scan_geometry/include" "${SLAM_SOURCE}" "${SLAM_SOURCE}/lib/karto_sdk/include" "${SLAM_SOURCE}/include")
ament_target_dependencies(coordinator_probe rclcpp std_srvs pluginlib)
target_link_libraries(coordinator_probe Ceres::ceres Boost::serialization Boost::thread Boost::filesystem Boost::system TBB::tbb)
# Test-only access permits calling the private virtual base cache rebuild from
# its throwing subclass. Production code has no fault-injection branch or flag.
target_compile_options(coordinator_probe PRIVATE -O1 -g0 -fno-access-control -fsanitize=undefined,float-cast-overflow -fno-sanitize-recover=all)
target_link_options(coordinator_probe PRIVATE -fsanitize=undefined,float-cast-overflow -fno-sanitize-recover=all)
